### 1.2 改造目标
**核心创新点：**
1. **轻量级模型替换**：GPT-4 → Qwen-2-0.5B/1.8B + 量化部署
2. **RAG 增强**：构建 PostgreSQL 专属知识库 + Sentence-BERT + FAISS
3. **场景感知**：三类场景识别（高并发 OLTP、批量 OLAP、混合负载）
4. **安全机制**：灰度生效 + 参数回滚 + 规则引擎
5. **历史上下文**：调优历史管理 + 迭代优化

```python
# src/rag_engine/knowledge_builder.py
class KnowledgeBuilder:
    def __init__(self):
        self.encoder = SentenceTransformer('paraphrase-MiniLM-L6-v2')  # 轻量级编码器
        self.index = None
        
    def build_knowledge_base(self):
        """
        1. 解析现有 knowledge_collection/postgres/
        2. 提取关键信息（参数、场景、建议值）
        3. 按场景分类组织
        """
        
    def create_embeddings(self, documents):
        """使用 Sentence-BERT 生成向量"""
        embeddings = self.encoder.encode(documents)
        return embeddings
        
    def build_faiss_index(self, embeddings):
        """构建 FAISS 索引"""
        dimension = embeddings.shape[1]
        index = faiss.IndexFlatL2(dimension)
        index.add(embeddings)
        return index
```

#### 2.1.2 RAG 检索器
```python
# src/rag_engine/retriever.py
class RAGRetriever:
    def __init__(self, knowledge_base, faiss_index, encoder):
        self.kb = knowledge_base
        self.index = faiss_index
        self.encoder = encoder
        
    def retrieve(self, query, scenario, top_k=5):
        """
        根据查询和场景检索相关知识
        Args:
            query: 当前问题（如"推荐 shared_buffers 值"）
            scenario: 场景类型（OLTP/OLAP/HYBRID）
            top_k: 返回前 k 条结果
        """
        # 1. 场景过滤
        filtered_docs = self.kb.filter_by_scenario(scenario)
        
        # 2. 向量检索
        query_vector = self.encoder.encode([query])
        distances, indices = self.index.search(query_vector, top_k)
        
        # 3. 返回上下文
        context = [filtered_docs[idx] for idx in indices[0]]
        return context
```

**量化部署：**
```python
# src/llm_engine/qwen_adapter.py
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

class QwenAdapter:
    def __init__(self, model_name="Qwen/Qwen2-0.5B-Instruct", use_quantization=True):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        
        if use_quantization:
            # 使用 4-bit 量化降低内存占用
            self.model = AutoModelForCausalLM.from_pretrained(
                model_name,
                load_in_4bit=True,
                device_map="auto"
            )
        else:
            self.model = AutoModelForCausalLM.from_pretrained(model_name)
            
    def generate(self, prompt, max_length=512):
        inputs = self.tokenizer(prompt, return_tensors="pt")
        outputs = self.model.generate(
            **inputs,
            max_length=max_length,
            temperature=0.7,
            do_sample=True
        )
        return self.tokenizer.decode(outputs[0], skip_special_tokens=True)
```

#### 2.2.2 替换 GPT 调用
**修改策略：**
```python
# src/knowledge_handler/llm_wrapper.py
class LLMWrapper:
    """统一 LLM 接口，支持 GPT 和 Qwen 切换"""
    def __init__(self, llm_type="qwen", **kwargs):
        if llm_type == "qwen":
            from llm_engine.qwen_adapter import QwenAdapter
            self.llm = QwenAdapter(**kwargs)
        elif llm_type == "openai":
            from knowledge_handler.gpt import GPT
            self.llm = GPT(**kwargs)
            
    def get_response(self, prompt, use_rag=True, scenario=None):
        """
        获取 LLM 响应
        Args:
            prompt: 原始提示
            use_rag: 是否使用 RAG 增强
            scenario: 场景类型
        """
        if use_rag and hasattr(self, 'rag_retriever'):
            # RAG 增强
            context = self.rag_retriever.retrieve(prompt, scenario)
            enhanced_prompt = self._build_rag_prompt(prompt, context)
            return self.llm.generate(enhanced_prompt)
        else:
            return self.llm.generate(prompt)
```

```python
class ScenarioClassifier:
    """
    基于负载特征自动识别场景类型
    """
    def classify(self, workload_stats):
        """
        Args:
            workload_stats: {
                'qps': 1000,
                'avg_query_time': 0.05,
                'read_write_ratio': 0.8,
                'query_complexity': 'simple',  # simple/complex
                'transaction_rate': 500
            }
        Returns:
            scenario: 'OLTP' | 'OLAP' | 'HYBRID'
        """
        # 规则引擎判断
        if workload_stats['qps'] > 500 and workload_stats['query_complexity'] == 'simple':
            return 'OLTP'
        elif workload_stats['avg_query_time'] > 1.0 and workload_stats['query_complexity'] == 'complex':
            return 'OLAP'
        else:
            return 'HYBRID'
```

### 2.4 规则引擎与安全机制

#### 2.4.1 参数冲突检测
```python
# src/rule_engine/conflict_detector.py
class ConflictDetector:
    """检测参数配置冲突"""
    
    CONFLICT_RULES = [
        {
            'params': ['shared_buffers', 'effective_cache_size'],
            'rule': lambda sb, ecs: ecs >= sb,
            'message': 'effective_cache_size 应 >= shared_buffers'
        },
        {
            'params': ['max_connections', 'work_mem'],
            'rule': lambda mc, wm: mc * wm < get_total_memory() * 0.8,
            'message': 'max_connections * work_mem 不应超过总内存 80%'
        }
    ]
    
    def check(self, config):
        """检查配置是否存在冲突"""
        conflicts = []
        for rule in self.CONFLICT_RULES:
            values = [config.get(p) for p in rule['params']]
            if not rule['rule'](*values):
                conflicts.append(rule['message'])
        return conflicts
```

#### 2.4.2 灰度生效与回滚
```python
# src/rule_engine/safe_applier.py
class SafeConfigApplier:
    """安全配置应用器"""
    
    def __init__(self, dbms):
        self.dbms = dbms
        self.backup_config = None
        
    def apply_with_rollback(self, new_config, test_duration=60):
        """
        灰度应用配置，支持自动回滚
        
        Args:
            new_config: 新配置
            test_duration: 测试时长（秒）
        """
        # 1. 备份当前配置
        self.backup_config = self.dbms.get_current_config()
        
        # 2. 应用新配置
        try:
            self.dbms.apply_config(new_config)
            
            # 3. 监控测试
            is_stable = self._monitor_stability(test_duration)
            
            if not is_stable:
                raise Exception("配置不稳定，触发回滚")
                
            return True, "配置成功应用"
            
        except Exception as e:
            # 4. 回滚
            self.rollback()
            return False, f"应用失败：{str(e)}"
            
    def rollback(self):
        """回滚到备份配置"""
        if self.backup_config:
            self.dbms.apply_config(self.backup_config)
            print("已回滚至备份配置")
```

### 2.5 历史上下文管理

```python
# src/history_manager/context_manager.py
class HistoryContextManager:
    """管理历史调优数据，支持迭代优化"""
    
    def __init__(self, db_path="./history_tuning.db"):
        self.conn = sqlite3.connect(db_path)
        self._init_db()
        
    def _init_db(self):
        """初始化数据库"""
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS tuning_history (
                id INTEGER PRIMARY KEY,
                scenario TEXT,
                config JSON,
                performance_metrics JSON,
                timestamp DATETIME,
                success BOOLEAN
            )
        """)
        
    def record(self, scenario, config, metrics, success):
        """记录调优结果"""
        self.conn.execute("""
            INSERT INTO tuning_history 
            (scenario, config, performance_metrics, timestamp, success)
            VALUES (?, ?, ?, datetime('now'), ?)
        """, (scenario, json.dumps(config), json.dumps(metrics), success))
        self.conn.commit()
        
    def get_successful_cases(self, scenario, top_k=5):
        """获取成功案例"""
        cursor = self.conn.execute("""
            SELECT config, performance_metrics 
            FROM tuning_history
            WHERE scenario = ? AND success = 1
            ORDER BY timestamp DESC
            LIMIT ?
        """, (scenario, top_k))
        return cursor.fetchall()
```
